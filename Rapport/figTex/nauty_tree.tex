
% Encircle a target cell inline.
\newcommand{\tc}[1]{%
\tikz[baseline= (c.base)]{
\node[
    draw,
    ellipse,
    inner sep=1.5pt,
    outer sep=0pt,
    line width=0.4pt
] (c) {#1};
}%
}


\newcommand{\ktgraph}[3]{%

    scale=0.42,
    every node/.style={
        circle,
        draw,
        inner sep=0pt,
        minimum size=6mm,
        font=\tiny,
        line width=0.5pt
    }
]

\draw (g0)--(g1);
\draw (g0)--(g2);
\draw (g0)--(g3);

\end{tikzpicture}%
}
